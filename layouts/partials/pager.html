{{/* Chapter pager — shown at the bottom of every book page. Finds the current page's position in the weight-sorted list of all pages under the current section and emits prev/next links. */}} {{ $pages := sort .CurrentSection.Pages "Params.weight" "asc" }} {{ $cur := .RelPermalink }} {{ $idx := -1 }} {{ range $i, $p := $pages }} {{ if eq $p.RelPermalink $cur }}{{ $idx = $i }}{{ end }} {{ end }} {{ if ge $idx 0 }}
{{ if gt $idx 0 }} {{ $prev := index $pages (sub $idx 1) }} ← Previous {{ $prev.Title }} {{ end }}
{{ if lt $idx (sub (len $pages) 1) }} {{ $next := index $pages (add $idx 1) }} Next → {{ $next.Title }} {{ end }}
{{ end }}
